Drop-in components for the current root-only Vue surface.

## Load DOCX bytes

```vue
<script setup lang="ts">
import { ref, watchEffect } from 'vue';
import { DocxEditor } from '@docx-editor.dev/vue';

const props = defineProps<{ url: string }>();
const doc = ref<Uint8Array>();

watchEffect(async () => {
  doc.value = new Uint8Array(await fetch(props.url).then((r) => r.arrayBuffer()));
});
</script>

<template>
  <DocxEditor :document="doc" />
</template>
```

## Save through the template ref

```vue
<script setup lang="ts">
import { useTemplateRef } from 'vue';
import { DocxEditor, type DocxEditorRef } from '@docx-editor.dev/vue';

const props = defineProps<{ docId: string; bytes: Uint8Array }>();
const editor = useTemplateRef<DocxEditorRef>('editor');

async function save() {
  const buf = await editor.value?.save();
  if (buf) {
    await fetch(`/api/documents/${props.docId}`, { method: 'PUT', body: buf });
  }
}
</script>

<template>
  <button @click="save">Save</button>
  <DocxEditor ref="editor" :document="props.bytes" />
</template>
```

## Automate through the server runtime

```ts
import { DocxEditor } from '@docx-editor.dev/editor-api';

export async function summarize(bytes: Uint8Array) {
  const runtime = await DocxEditor.createServer(bytes, { author: 'Assistant' });
  try {
    return await runtime.run(async (context) => {
      const body = context.document.body;
      body.load('text');
      await context.sync();
      return body.text;
    });
  } finally {
    runtime.dispose();
  }
}
```
