Import `@docx-editor.dev/vue/styles.css` once in your application entry. Give each editor a container with a height.

## Load DOCX bytes

```vue
<script setup lang="ts">
import { DocxEditor, useDocxSource } from '@docx-editor.dev/vue';

const { document, error } = useDocxSource('/template.docx');
</script>

<template>
  <p v-if="error" role="alert">{{ error.message }}</p>
  <div v-else style="height: 100vh">
    <DocxEditor :document="document" title="Proposal.docx" />
  </div>
</template>
```

## Save through the ref

```vue
<script setup lang="ts">
import { ref } from 'vue';
import { DocxEditor, type DocxEditorRef } from '@docx-editor.dev/vue';

defineProps<{ bytes: Uint8Array }>();
const editorRef = ref<DocxEditorRef | null>(null);

async function save() {
  const next = await editorRef.value?.save();
  if (!next) return;
  await fetch('/api/documents/42', { method: 'PUT', body: next });
}
</script>

<template>
  <button @click="save">Save</button>
  <DocxEditor ref="editorRef" :document="bytes" />
</template>
```

## Compose custom chrome

Create a button that uses the editor context:

```vue
<script setup lang="ts">
import { useEditorCommand } from '@docx-editor.dev/vue';

const bold = useEditorCommand('text.bold');
</script>

<template>
  <button
    @mousedown.prevent
    :disabled="!bold.isEnabled.value"
    :aria-pressed="bold.isActive.value"
    :title="bold.disabledReason.value ?? 'Bold'"
    @click="bold.execute()"
  >
    Bold
  </button>
</template>
```

Then place the button and editor parts under the root:

```vue
<script setup lang="ts">
import { DocxEditor } from '@docx-editor.dev/vue';
import BoldButton from './BoldButton.vue';

defineProps<{ bytes: Uint8Array }>();
</script>

<template>
  <div class="docx-editor" style="height: 100vh; display: flex; flex-direction: column">
    <DocxEditor.Root :document="bytes">
      <DocxEditor.Toolbar>
        <BoldButton />
      </DocxEditor.Toolbar>
      <DocxEditor.Viewport style="flex: 1; min-height: 0">
        <DocxEditor.Navigation />
        <DocxEditor.Content />
        <DocxEditor.HyperLink />
        <DocxEditor.ContextMenu />
      </DocxEditor.Viewport>
    </DocxEditor.Root>
  </div>
</template>
```

`DocxEditor.Navigation` shares the viewport with the document and stays in place while the pages scroll. `DocxEditor.HyperLink` and `DocxEditor.ContextMenu` provide overlay chrome.

## Automate an open editor

Render this component as a descendant of `DocxEditor.Root`:

```vue
<script setup lang="ts">
import { useDocxEditor } from '@docx-editor.dev/vue';
import { DocxEditor as AutomationEditor } from '@docx-editor.dev/editor-api/browser';

const editor = useDocxEditor();

async function inspectFirstParagraph() {
  if (!editor.value) return;
  const runtime = AutomationEditor.createBrowser(editor.value);

  try {
    await runtime.run(async (context) => {
      const first = context.document.body.paragraphs.getFirst();
      first.load('text');
      await context.sync();
      console.log(first.text);
    });
  } finally {
    runtime.dispose();
  }
}
</script>

<template>
  <button :disabled="!editor" @click="inspectFirstParagraph">Inspect first paragraph</button>
</template>
```

Install `@docx-editor.dev/editor-api` before you use this example. The runtime uses the editor that `DocxEditor.Root` created.

## Fetch bytes and fonts together

Install `@docx-editor.dev/fonts` to use the packaged font resolver:

```bash
npm install @docx-editor.dev/fonts
```

Load document bytes with fonts for measurement:

```vue
<script setup lang="ts">
import {
  DocxEditorRoot,
  DocxEditorViewport,
  DocxEditorContent,
  useDocxSource,
} from '@docx-editor.dev/vue';
import { packagedFonts } from '@docx-editor.dev/fonts';

const { document, fonts, error, isLoading } = useDocxSource('/sample.docx', {
  fonts: packagedFonts(),
});
</script>

<template>
  <div v-if="document" style="height: 100vh">
    <DocxEditorRoot :document="document" :fonts="fonts">
      <DocxEditorViewport style="height: 100%">
        <DocxEditorContent />
      </DocxEditorViewport>
    </DocxEditorRoot>
  </div>
  <p v-else-if="error">{{ error.message }}</p>
  <p v-else-if="isLoading" aria-live="polite">Loading document</p>
</template>
```
