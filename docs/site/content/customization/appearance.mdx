Set `--doc-*` properties on your `.docx-editor` wrapper. Load application CSS after the editor stylesheet.

Add the `dark` class to a composed wrapper for dark controls. Use `colorMode` on the packaged editor. Dark mode also adjusts the document preview. Saved document colors remain unchanged.

## Style your controls

Keep CSS under your wrapper. These tokens and selectors work in both frameworks.

```css
.docx-editor.contract-workspace {
  --doc-dialog-radius: 16px;
  --doc-dialog-control-radius: 8px;
  --doc-dialog-padding: 24px;
}
.docx-editor.contract-workspace .contract-toolbar {
  margin: 0;
  border-radius: 0;
  padding: 8px 12px;
  background: var(--doc-surface);
  border-bottom: 1px solid var(--doc-border);
}
.docx-editor.contract-workspace .contract-card {
  border-radius: 12px;
  background: var(--doc-surface);
}
```

## React and Vue reference

Pass your DOCX bytes and loaded fonts to this component. The preview uses a fictional contract.

<FrameworkTabs variant="block">
<Framework value="react">

```tsx
import { DocxEditor, type DocxEditorRootProps } from '@docx-editor.dev/react';
import { reviewModule, DocxEditorReview } from '@docx-editor.dev/pro/react';
import '@docx-editor.dev/core/styles/editor.css';

const modules = [reviewModule()];

export function ContractEditor({
  bytes,
  fonts,
}: {
  bytes: Uint8Array;
  fonts: DocxEditorRootProps['fonts'];
}) {
  return (
    <div className="docx-editor contract-workspace">
      <DocxEditor.Root
        document={bytes}
        fonts={fonts}
        modules={modules}
        author="Demo reviewer"
        mode="suggesting"
        reviewDisplayMode="all-markup"
      >
        <DocxEditor.Toolbar className="contract-toolbar" />
        <DocxEditor.Viewport style={{ height: 520 }}>
          <DocxEditor.Content />
          <DocxEditor.HyperLink />
          <DocxEditor.ContextMenu />
          <DocxEditorReview />
        </DocxEditor.Viewport>
      </DocxEditor.Root>
    </div>
  );
}
```

</Framework>
